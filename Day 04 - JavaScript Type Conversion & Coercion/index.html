1️⃣ Explicit Conversion (We do it manually) let num = "100"; console.log(Number(num)); // 100 console.log(String(100)); // "100" console.log(Boolean(0)); // false console.log(Boolean("hello")); // true 2️⃣ Implicit Coercion (JS does it automatically) console.log("5" + 1); // "51" (number becomes string) console.log("5" - 1); // 4 (string becomes number) console.log("5" * "2"); // 10 (both become numbers) console.log(1 + true); // 2 (true becomes 1) ⚠️ Common JavaScript Gotcha console.log("5" + 3); // "53" console.log("5" - 3); // 2 + with a string triggers concatenation, but - * /  force numeric conversion.
